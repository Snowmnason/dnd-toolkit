/**
 * Shared types for get_feature_flags Edge Function
 * These types mirror the database schema and client-side types
 */

export type OverrideTargetType = "flag" | "entitlement";

/**
 * Feature flag row from database
 */
export interface FeatureFlagRow {
  flag_name: string;
  enabled: boolean;
  /** Optional list of dependent flag names */
  depends_on?: string[] | null;
  /** Advanced condition tree (Phase 3) stored as JSON */
  condition_logic?: Record<string, any> | null;
  /** Optional structured metadata for nested configurations (e.g., logger categories) */
  metadata?: Record<string, any> | null;
  kind: string;
  description?: string;
  created_at: string;
  updated_at: string;
}

/**
 * Entitlement row from database
 * Includes new 003 schema columns: is_active (soft-delete flag), remind_user (expiry reminder)
 */
export interface EntitlementRow {
  id: string;
  user_id: string;
  key: string;
  is_active: boolean; // false = marked inactive by background job or manual revocation
  remind_user: boolean; // true = user should be reminded before expiry
  created_at: string;
  updated_at: string;
  expires_at: string | null;
}

/**
 * Cohort definition
 */
export interface CohortRow {
  id: string;
  slug: string;
  name: string;
  description?: string | null;
  percentage: number;
  seed?: string | null;
  is_active: boolean;
  metadata?: Record<string, any> | null;
  created_at: string;
  updated_at: string;
}

/**
 * Cohort <-> Flag assignment row
 */
export interface CohortFlagAssignmentRow {
  id: string;
  flag_name: string;
  cohort_id: string;
  enabled: boolean;
  created_at: string;
  updated_at: string;
  cohort_slug?: string | null; // populated by edge function for client convenience
}

/**
 * Explicit user cohort membership
 */
export interface UserCohortMembershipRow {
  id: string;
  user_id: string;
  cohort_id: string;
  source: string; // e.g., 'admin' | 'system'
  created_by?: string | null;
  reason?: string | null;
  expires_at?: string | null;
  created_at: string;
  updated_at: string;
  is_active?: boolean;
  cohort_slug?: string | null; // populated by edge function for client convenience
}

/**
 * Feature flag override row from database (003 simplified schema, no target_type in DB)
 * target_type added synthetically by Edge Function for client compatibility
 */
export interface FeatureFlagOverrideRow {
  id: string;
  user_id: string;
  target_type: "flag"; // Synthetic: identifies this as a feature flag override
  target_name: string; // Feature flag name (from feature_flags.flag_name)
  enabled: boolean;
  expires_at: string | null;
  revoked: boolean;
  reason?: string;
  created_by?: string | null;
  created_at: string;
  updated_at: string;
}

/**
 * Entitlement override row from database (NEW in 003: entitlements_overrides table)
 * Allows admins to manually grant/revoke entitlements independent of expiry logic
 */
export interface EntitlementOverrideRow {
  id: string;
  user_id: string;
  target_type: "entitlement"; // Synthetic: identifies this as an entitlement override
  target_name: string; // Entitlement key (from feature_flags.entitlements.key)
  action: "grant" | "revoke"; // Override action
  expires_at: string | null; // Override expiry (independent of base entitlement)
  revoked: boolean; // Soft-delete flag
  reason?: string; // Admin reason for override
  created_by?: string | null; // Admin who created override
  created_at: string;
  updated_at: string;
}

/**
 * Rollout configuration for gradual feature rollouts
 * Returned by Edge Function alongside flags/entitlements/overrides
 */
export interface RolloutConfigRow {
  percentage: number; // 0-100
  seed?: string; // Optional seed for rebalancing (e.g., "2026-02-07")
}

/**
 * Response structure from Edge Function
 * Client expects this exact structure for caching and merge logic
 * Overrides array includes both flag and entitlement overrides (differentiated by target_type)
 */
export interface GetFeatureFlagsResponse {
  flags: FeatureFlagRow[];
  entitlements: EntitlementRow[];
  overrides: (FeatureFlagOverrideRow | EntitlementOverrideRow)[]; // Both flag and entitlement overrides (non-revoked, non-expired)
  rollouts: Record<string, RolloutConfigRow>; // Rollout config by flag name
  cohorts?: CohortRow[]; // Active cohort definitions
  cohort_assignments?: CohortFlagAssignmentRow[]; // Mapping of cohorts -> flags
  user_cohort_memberships?: UserCohortMembershipRow[]; // Explicit memberships for requesting user
  fetchedAt: number;
  version: "v1";
}

/**
 * JWT claims from Supabase auth token
 */
export interface JWTClaims {
  sub: string; // User ID
  email?: string;
  aud?: string;
  iat?: number;
  exp?: number;
  [key: string]: unknown;
}
