/**
 * Server-side query helpers for get_feature_flags Edge Function
 * These functions are called server-side to fetch data from Supabase
 * Moved from client-side lib/database/feature-flags.ts, lib/database/entitlements.ts, lib/database/feature-flag-overrides.ts
 */

import { type SupabaseClient } from "npm:@supabase/supabase-js@2.39.0";
import type {
  EntitlementOverrideRow,
  EntitlementRow,
  FeatureFlagOverrideRow,
  FeatureFlagRow,
} from "./types.ts";

/**
 * Fetch all global feature flags
 * Moved from lib/database/feature-flags.ts
 *
 * @param supabase - Already initialized Supabase client (created once per request)
 * @returns List of all feature flags in the system
 */
export async function fetchFeatureFlags(
  supabase: SupabaseClient,
): Promise<FeatureFlagRow[]> {
  const { data, error } = await supabase
    .schema('feature_flags')
    .from('feature_flags')
    .select(
      "flag_name, enabled, kind, description, created_at, updated_at, depends_on, condition_logic, metadata",
    );

  if (error) {
    console.error("Error fetching feature flags:", error?.message ?? error);
    throw new Error(
      `Failed to fetch feature flags: ${error?.message ?? String(error)}`,
    );
  }

  return (data || []) as FeatureFlagRow[];
}

/**
 * Fetch all entitlements for a specific user (server-side filtered for active)
 * Moved from lib/database/entitlements.ts
 *
 * Server-side filters:
 * - is_active = true (excludes soft-deleted/revoked entitlements marked by background job or admin)
 * - non-expired (expires_at IS NULL OR expires_at > now())
 *
 * Includes new 003 schema columns: is_active, remind_user
 *
 * @param supabase - Already initialized Supabase client (created once per request)
 * @param userId - User ID (UUID) to fetch entitlements for
 * @returns List of active user entitlements (is_active=true, non-expired only)
 */
export async function fetchEntitlementsByUserId(
  supabase: SupabaseClient,
  userId: string,
): Promise<EntitlementRow[]> {
  const now = new Date().toISOString();

  const { data, error } = await supabase
    .schema('feature_flags')
    .from('entitlements')
    .select(
      "id, user_id, key, is_active, remind_user, created_at, updated_at, expires_at",
    )
    .eq("user_id", userId)
    .eq("is_active", true)
    .or(`expires_at.is.null,expires_at.gt.${now}`);

  if (error) {
    console.error(
      `Error fetching entitlements for user ${userId}:`,
      error?.message ?? error,
    );
    throw new Error(
      `Failed to fetch entitlements for user ${userId}: ${error?.message ?? String(error)}`,
    );
  }

  return (data || []) as EntitlementRow[];
}

/**
 * Fetch all active overrides for feature flags for a specific user
 * Moved from lib/database/feature-flag-overrides.ts
 *
 * 003 Schema Change: Removed target_type column from database (simplified to single table for flags).
 * target_type is now set synthetically by Edge Function to "flag" for client compatibility.
 *
 * Server-side filters for non-revoked, non-expired overrides.
 *
 * @param supabase - Already initialized Supabase client (created once per request)
 * @param userId - User ID (UUID) to fetch overrides for
 * @returns List of active feature flag overrides (non-revoked, non-expired) with synthetic target_type="flag"
 */
export async function fetchOverridesByUserId(
  supabase: SupabaseClient,
  userId: string,
): Promise<FeatureFlagOverrideRow[]> {
  const now = new Date().toISOString();

  const { data, error } = await supabase
    .schema('feature_flags')
    .from('feature_flag_overrides')
    .select(
      "id, user_id, flag_name, enabled, expires_at, revoked, reason, created_by, created_at, updated_at",
    )
    .eq("user_id", userId)
    .eq("revoked", false)
    .or(`expires_at.is.null,expires_at.gt.${now}`);

  if (error) {
    console.error(
      `Error fetching overrides for user ${userId}:`,
      error?.message ?? error,
    );
    throw new Error(
      `Failed to fetch feature flag overrides for user ${userId}: ${error?.message ?? String(error)}`,
    );
  }

  // Add synthetic fields for client compatibility:
  // - target_type: always "flag" (distinguishes from entitlement overrides)
  // - target_name: maps DB column flag_name → client-expected field
  return (data || []).map((row: any) => ({
    ...row,
    target_type: "flag" as const,
    target_name: row.flag_name,
  })) as FeatureFlagOverrideRow[];
}

/**
 * Fetch all active overrides for entitlements for a specific user
 * NEW in 003: Separate table for admin overrides (grant/revoke) on entitlements
 *
 * Server-side filters for non-revoked, non-expired overrides.
 * Allows admins to temporarily grant/revoke entitlements outside normal expiry logic.
 *
 * @param supabase - Already initialized Supabase client (created once per request)
 * @param userId - User ID (UUID) to fetch entitlement overrides for
 * @returns List of active entitlement overrides (non-revoked, non-expired) with synthetic target_type="entitlement"
 */
export async function fetchEntitlementOverridesByUserId(
  supabase: SupabaseClient,
  userId: string,
): Promise<EntitlementOverrideRow[]> {
  const now = new Date().toISOString();

  const { data, error } = await supabase
    .schema('feature_flags')
    .from('entitlements_overrides')
    .select(
      "id, user_id, entitlement_key, is_active, expires_at, revoked, reason, created_by, created_at, updated_at",
    )
    .eq("user_id", userId)
    .eq("revoked", false)
    .or(`expires_at.is.null,expires_at.gt.${now}`);

  if (error) {
    console.error(
      `Error fetching entitlement overrides for user ${userId}:`,
      error?.message ?? error,
    );
    throw new Error(
      `Failed to fetch entitlement overrides for user ${userId}: ${error?.message ?? String(error)}`,
    );
  }

  // Map DB columns to client-expected synthetic fields:
  // - target_type: always "entitlement" (distinguishes from flag overrides)
  // - target_name: maps DB column entitlement_key → client-expected field
  // - action: maps DB boolean is_active → "grant"/"revoke" string for client
  // - enabled: maps DB boolean is_active → boolean for client compatibility
  return (data || []).map((row: any) => ({
    id: row.id,
    user_id: row.user_id,
    target_type: "entitlement" as const,
    target_name: row.entitlement_key,
    action: row.is_active ? "grant" : "revoke",
    expires_at: row.expires_at,
    revoked: row.revoked,
    reason: row.reason,
    created_by: row.created_by,
    created_at: row.created_at,
    updated_at: row.updated_at,
    enabled: row.is_active, // Synthetic: maps boolean to enabled/disabled for client compatibility
  })) as EntitlementOverrideRow[];
}

/**
 * NEW: Fetch all active rollout configurations
 * Server-side filtered to only return active rollouts (is_active = true)
 *
 * @param supabase - Already initialized Supabase client (created once per request)
 * @returns Map of rollout configs by flag name { flagName: { percentage, seed? } }
 */
export async function fetchRolloutConfigs(
  supabase: SupabaseClient,
): Promise<Record<string, { percentage: number; seed?: string }>> {
  const { data, error } = await supabase
    .schema('feature_flags')
    .from('feature_flag_rollouts')
    .select("flag_name, percentage, seed")
    .eq("is_active", true);

  if (error) {
    console.error("Error fetching rollout configs:", error?.message ?? error);
    throw new Error(
      `Failed to fetch rollout configs: ${error?.message ?? String(error)}`,
    );
  }

  // Convert array to map for fast lookups
  const rollouts: Record<string, { percentage: number; seed?: string }> = {};
  if (data && data.length > 0) {
    for (const row of data) {
      rollouts[row.flag_name] = {
        percentage: row.percentage,
        seed: row.seed || undefined,
      };
    }
  }

  return rollouts;
}
