/**
 * Health Check Edge Function
 *
 * Public endpoint for network availability checks.
 * No authentication required - returns 200 OK if the system is operational.
 *
 * Used by: Network detection code to verify connectivity without auth errors.
 * Prevents 401 noise from unauthenticated pings.
 *
 * **Request:**
 * ```
 * HEAD or GET /functions/v1/health
 * ```
 *
 * **Response (Success):**
 * ```
 * HTTP 200 OK
 * {
 *   "status": "ok",
 *   "timestamp": 1707230400000,
 *   "uptime": 3600000,
 *   "version": "1.0"
 * }
 * ```
 *
 * **Response (Maintenance):**
 * ```
 * HTTP 503 Service Unavailable
 * {
 *   "status": "maintenance",
 *   "message": "System is under maintenance"
 * }
 * ```
 */

const startTime = Date.now();
const VERSION = Deno.env.get("HEALTH_VERSION") ?? "1.0";

Deno.serve(async (req: Request) => {
  // Handle OPTIONS preflight requests
  if (req.method === "OPTIONS") {
    return new Response(null, {
      status: 204,
      headers: {
        Allow: "GET, HEAD, OPTIONS",
      },
    });
  }

  // Support both HEAD and GET requests
  if (req.method !== "GET" && req.method !== "HEAD") {
    return json(
      {
        error: "Method not allowed",
        allowed: ["GET", "HEAD"],
      },
      405,
      { Allow: "GET, HEAD, OPTIONS" },
    );
  }

  try {
    // Check for maintenance mode (optional environment variable)
    const maintenanceMode = Deno.env.get("MAINTENANCE_MODE") === "true";
    if (maintenanceMode) {
      const maintenanceResponse = {
        status: "maintenance",
        message: "System is under maintenance",
        timestamp: Date.now(),
      };
      // Return empty body for HEAD requests
      if (req.method === "HEAD") {
        return new Response(null, {
          status: 503,
          headers: getResponseHeaders(),
        });
      }
      return json(maintenanceResponse, 503);
    }

    // Return health status
    const response = {
      status: "ok",
      timestamp: Date.now(),
      uptime: Date.now() - startTime,
      version: VERSION,
    };

    // Return empty body for HEAD requests (headers only)
    if (req.method === "HEAD") {
      return new Response(null, {
        status: 200,
        headers: getResponseHeaders(),
      });
    }

    return json(response, 200);
  } catch (error) {
    const errorMsg = error instanceof Error ? error.message : String(error);
    console.error("Health check error:", errorMsg);

    // Do not expose internal error details to clients
    const errorResponse = {
      status: "error",
      message: "Health check failed",
      timestamp: Date.now(),
    };

    // Return empty body for HEAD requests
    if (req.method === "HEAD") {
      return new Response(null, {
        status: 500,
        headers: getResponseHeaders(),
      });
    }

    return json(errorResponse, 500);
  }
});

/**
 * Get response headers for health check responses
 */
function getResponseHeaders(): Record<string, string> {
  return {
    "Content-Type": "application/json; charset=utf-8",
    "Cache-Control": "no-cache, no-store, must-revalidate",
    Pragma: "no-cache",
    Expires: "0",
  };
}

/**
 * Helper to return JSON responses
 * Supports optional headers parameter
 */
function json(
  data: any,
  status: number = 200,
  headers: Record<string, string> = {},
): Response {
  return new Response(JSON.stringify(data), {
    status,
    headers: {
      ...getResponseHeaders(),
      ...headers,
    },
  });
}
