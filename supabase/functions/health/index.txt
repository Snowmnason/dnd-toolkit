// Health Check Edge Function
const startTime = Date.now();
const VERSION = Deno.env.get("HEALTH_VERSION") ?? "1.0";

// Common headers returned for JSON responses and browser checks
const COMMON_HEADERS = {
  "Content-Type": "application/json; charset=utf-8",
  "Cache-Control": "no-cache, no-store, must-revalidate",
  Pragma: "no-cache",
  Expires: "0",
  // CORS - allow lightweight checks from monitoring/origin pages if needed
  "Access-Control-Allow-Origin": "*",
  "Access-Control-Allow-Methods": "GET, HEAD, OPTIONS",
  "Access-Control-Allow-Headers": "Authorization, Content-Type",
};

Deno.serve(async (req: Request) => {
  const url = new URL(req.url);
  const method = req.method.toUpperCase();

  // OPTIONS preflight
  if (method === "OPTIONS") {
    return new Response(null, {
      status: 204,
      headers: COMMON_HEADERS,
    });
  }

  // Only allow GET and HEAD
  if (method !== "GET" && method !== "HEAD") {
    return new Response(JSON.stringify({
      error: "Method not allowed",
      allowed: ["GET", "HEAD"],
    }), {
      status: 405,
      headers: {
        ...COMMON_HEADERS,
        Allow: "GET, HEAD, OPTIONS",
      },
    });
  }

  try {
    const maintenanceMode = Deno.env.get("MAINTENANCE_MODE") === "true";
    if (maintenanceMode) {
      const maintenancePayload = {
        status: "maintenance",
        message: "System is under maintenance",
        timestamp: Date.now(),
        version: VERSION,
      };

      // HEAD returns headers with lightweight health info
      if (method === "HEAD") {
        return new Response(null, {
          status: 503,
          headers: {
            ...COMMON_HEADERS,
            "X-Health-Status": maintenancePayload.status,
            "X-Health-Version": String(maintenancePayload.version),
            "X-Health-Timestamp": String(maintenancePayload.timestamp),
          },
        });
      }

      return new Response(JSON.stringify(maintenancePayload), {
        status: 503,
        headers: COMMON_HEADERS,
      });
    }

    const payload = {
      status: "ok",
      timestamp: Date.now(),
      uptime: Date.now() - startTime,
      version: VERSION,
    };

    // For HEAD requests provide concise health headers so callers that
    // prefer lightweight checks (curl -I or browser HEAD) can still inspect
    // health metadata without parsing a body. For GET return full JSON.
    if (method === "HEAD") {
      return new Response(null, {
        status: 200,
        headers: {
          ...COMMON_HEADERS,
          "X-Health-Status": payload.status,
          "X-Health-Version": String(payload.version),
          "X-Health-UptimeMs": String(payload.uptime),
          "X-Health-Timestamp": String(payload.timestamp),
        },
      });
    }

    return new Response(JSON.stringify(payload), {
      status: 200,
      headers: COMMON_HEADERS,
    });
  } catch (err) {
    console.error("Health check error:", err instanceof Error ? err.message : String(err));

    if (method === "HEAD") {
      return new Response(null, {
        status: 500,
        headers: COMMON_HEADERS,
      });
    }

    return new Response(JSON.stringify({
      status: "error",
      message: "Health check failed",
      timestamp: Date.now(),
    }), {
      status: 500,
      headers: COMMON_HEADERS,
    });
  }
});
